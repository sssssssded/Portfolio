``` Server  
#include <iostream>
#include <WinSock2.h>

// 소켓 라이브러리 사용.
#pragma comment(lib, "ws2_32.lib")

int main()
{
   WSADATA   wsaData;

   WSAStartup(MAKEWORD(2, 2), &wsaData);

   // ListenSocket을 생성한다.
   // SOCK_STREAM : 연결 지향형 소켓
   // SOCK_DGRAM : 비연결 지향형 소켓
   SOCKET   ListenSocket = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);

   // 위에서 생성한 ListenSocket에 주소를 부여한다.
   // 주소는 IP주소와 Port번호로 구성된다.
   // 내부 IP, 외부 IP
   SOCKADDR_IN   ListenAddr = {};

   // IPv4타입 주소 사용
   ListenAddr.sin_family = AF_INET;

   // 0x12345678
   // 리틀엔디안 : 78 56 34 12
   // 빅엔디안 : 12 34 56 78
   // 네트워크에서 포트는 빅엔디안을 사용한다.
   ListenAddr.sin_port = htons(10010);

   // 이 프로그램이 실행되는 장비의 IP로 세팅한다.
   ListenAddr.sin_addr.s_addr = INADDR_ANY;

   // 위에서 생성한 주소를 ListenSocket에 묶어준다.
   bind(ListenSocket, (SOCKADDR*)&ListenAddr, sizeof(ListenAddr));

   // ListenSocket이 접속요청을 하는 유저의 순서를 정할 큐를 만들고
   // 유저의 접속을 승인할 수 있는 상태로 만들어준다.
   listen(ListenSocket, SOMAXCONN);

   // 접속하는 유저를 승인할 수 있게 대기상태로 만든다.
   // 승인 되면 해당 유저와 연결된 소켓을 만들게 된다.
   SOCKADDR_IN   ClientAddr = {};
   int   ClientAddrSize = sizeof(ClientAddr);
   // accept는 유저가 접속할 때까지 대기상태가 된다. 즉 유저 접속 전에는 아래의 코드를
   // 동작시킬 수 없다.
   SOCKET ClientSocket = accept(ListenSocket, (SOCKADDR*)&ClientAddr, &ClientAddrSize);

   // ClientSocket을 이용해서 접속한 유저와 통신한다.
   // recv는 해당 소켓으로 전송되는 데이터를 받는다.
   // recv도 연결된 유저가 데이터를 보내기 전까지 대기상태가 된다.
   char   RecvData[256] = {};
   recv(ClientSocket, RecvData, 256, 0);

   std::cout << RecvData << std::endl;

   char   SendData[256] = {};
   strcpy_s(SendData, "데이터 리턴");
   send(ClientSocket, SendData, strlen(SendData), 0);

   // 소켓은 다 썼으면 반드시 제거해야 된다.
   closesocket(ClientSocket);
   closesocket(ListenSocket);


   WSACleanup();

   return 0;
}
```

```  MainThread Client

#include <iostream>
#include <string>
#include <thread>
#include <atomic>
#include <WinSock2.h>
#include <WS2tcpip.h>   // inet_pton

// 소켓 라이브러리 사용.
#pragma comment(lib, "ws2_32.lib")

constexpr const char* SERVER_IP = "127.0.0.1";   // 로컬
constexpr unsigned short SERVER_PORT = 10010;
constexpr int            BUFFER_SIZE = 256;

// 서버와 연결되어 있는지 여부.
std::atomic<bool> gConnected = true;

// 서버가 보내주는 다른 클라이언트의 메시지를 받아서 출력한다.
// 입력(getline)과 수신(recv)은 둘 다 대기하는 함수이므로 스레드를 나눠야 한다.
void RecvThread(SOCKET Socket)
{
    char RecvData[BUFFER_SIZE];

    while (true)
    {
        int Size = recv(Socket, RecvData, BUFFER_SIZE - 1, 0);

        // 0 : 서버가 연결을 끊음, 음수 : 오류 또는 내가 exit로 소켓을 닫음
        if (Size <= 0)
            break;

        RecvData[Size] = '\0';
        std::cout << RecvData << std::endl;
    }

    // 내가 exit를 입력한 게 아니라 서버쪽에서 끊긴 경우
    if (gConnected.exchange(false))
        std::cout << "서버와의 연결이 끊어졌습니다. 엔터를 누르면 종료됩니다." << std::endl;
}

int main()
{
    WSADATA wsaData;

    if (WSAStartup(MAKEWORD(2, 2), &wsaData) != 0)
    {
        std::cout << "WSAStartup 실패" << std::endl;
        return -1;
    }

    SOCKET Socket = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);

    if (Socket == INVALID_SOCKET)
    {
        std::cout << "socket 생성 실패 : " << WSAGetLastError() << std::endl;
        WSACleanup();
        return -1;
    }

    // 접속할 서버의 주소.
    SOCKADDR_IN ServerAddr = {};
    ServerAddr.sin_family = AF_INET;
    ServerAddr.sin_port = htons(SERVER_PORT);
    inet_pton(AF_INET, SERVER_IP, &ServerAddr.sin_addr);

    // connect : 서버에 접속 요청. 서버의 accept가 승인해주면 성공한다.
    if (connect(Socket, (SOCKADDR*)&ServerAddr, sizeof(ServerAddr)) == SOCKET_ERROR)
    {
        std::cout << "서버 접속 실패 (서버가 켜져 있는지 확인) : " << WSAGetLastError() << std::endl;
        closesocket(Socket);
        WSACleanup();
        return -1;
    }

    std::cout << "서버 접속 성공 : " << SERVER_IP << ":" << SERVER_PORT << std::endl;
    std::cout << "메시지를 입력하세요. (exit 입력 시 종료)" << std::endl;

    std::thread Recv(RecvThread, Socket);

    std::string Line;

    while (std::getline(std::cin, Line))
    {
        // 서버가 먼저 끊은 경우
        if (!gConnected)
            break;

        if (Line == "exit")
            break;

        if (Line.empty())
            continue;

        if (send(Socket, Line.c_str(), (int)Line.size(), 0) == SOCKET_ERROR)
        {
            std::cout << "전송 실패 : " << WSAGetLastError() << std::endl;
            break;
        }
    }

    // RecvThread에서 "연결 끊김" 메시지를 출력하지 않도록 먼저 false로 만든다.
    gConnected = false;

    // shutdown : 더 이상 주고받지 않겠다고 서버에 알린다.
    // 서버의 recv는 0을 반환하고, 이쪽 RecvThread의 recv도 깨어나서 빠져나온다.
    shutdown(Socket, SD_BOTH);

    Recv.join();

    closesocket(Socket);

    WSACleanup();

    std::cout << "클라이언트를 종료합니다." << std::endl;

    return 0;
}
```
